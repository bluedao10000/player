```sh
curl -fsS http://127.0.0.1:4310/work/naruto-tv-2002/chapter/kakashis-decision | rg -o 'id="passage-shared-lunch"'
curl -fsS http://127.0.0.1:4321/api/v1/knowledge/worlds/naruto-tv-2002/scenes/shared-lunch
```
